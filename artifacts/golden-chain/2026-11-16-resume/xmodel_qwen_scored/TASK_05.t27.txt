module coding_eval_gen;

pub fn square8(a: i8) i8 {
    return a * a;
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_square8() {
        assert_eq!(square8(-11), 121);
        assert_eq!(square8(0), 0);
        assert_eq!(square8(11), 121);
    }
}
test square8_works { assert square8(3) == 9; assert square8(-3) == 9; assert square8(0) == 0; assert square8(10) == 100; }
