module coding_eval_gen;

pub fn is_leap_word(y: i32) -> bool {
    if y % 4 == 0 {
        if y % 100 != 0 || y % 400 == 0 {
            return true;
        }
    }
    false
}
test is_leap_word_works { assert is_leap_word(2000); assert is_leap_word(2024); assert !is_leap_word(1900); assert !is_leap_word(2023); }
