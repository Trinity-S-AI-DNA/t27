module coding_eval_gen;

pub fn sum_to8(n: i8) i8 {
    let mut sum = 0;
    while n >= 0 {
        sum += n;
        n -= 1;
    }
    return sum;
}
test sum_to8_works { assert sum_to8(0) == 0; assert sum_to8(1) == 1; assert sum_to8(5) == 15; assert sum_to8(10) == 55; assert sum_to8(-3) == 0; }
