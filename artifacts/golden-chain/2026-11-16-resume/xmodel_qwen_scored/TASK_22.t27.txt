module coding_eval_gen;

pub fn clamp0_100(a: i8) i8 {
    if a < 0 {
        return 0;
    }
    if a > 100 {
        return 100;
    }
    return a as i8 / 100.0 * 100;
}
test clamp0_100_works { assert clamp0_100(-5) == 0; assert clamp0_100(42) == 42; assert clamp0_100(120) == 100; assert clamp0_100(0) == 0; assert clamp0_100(100) == 100; }
