module coding_eval_gen;

pub fn biggest_word(a: i32, b: i32) i32 {
    if a > b {
        return a;
    } else {
        return b;
    }
}
test biggest_word_works { assert biggest_word(1, 2) == 2; assert biggest_word(-9, -3) == -3; assert biggest_word(500000, 499999) == 500000; }
