module coding_eval_gen;

pub fn max3_8(a: i8, b: i8, c: i8) i8 {
    if a > b && a > c {
        return a;
    } else if b > a && b > c {
        return b;
    } else {
        return c;
    }
}
test max3_8_works { assert max3_8(1, 2, 3) == 3; assert max3_8(9, 2, 3) == 9; assert max3_8(-1, -2, -3) == -1; }
