module coding_eval_gen;

pub fn half_even8(a: i8) i8 {
    if a >= 0 && a <= 100 {
        return a / 2;
    } else {
        panic!("a must be between 0 and 100");
    }
}
test half_even8_works { assert half_even8(4) == 2; assert half_even8(0) == 0; assert half_even8(100) == 50; }
