// tb_diff_relu.v -- ONE common testbench for BOTH Verilog backends (t27c and
// t27b) of specs/ternary/gft_relu.t27. It does NOT use either compiler's own
// test blocks: the golden expectations below are computed independently from
// the GF-T16 encoding documented in the spec header.
//
// GF-T16 encoding in a u32 (per specs/ternary/gft_relu.t27 header and its
// passing test literals):
//   value bits  [15:0], sign bit [16]: bit16 set => negative,
//   so negative  <=>  (x >> 32'd16) == 32'd1   (x is u32; bits above 16 are 0)
//   +1.0 = 0x005000 (20480), +2.0 = 0x005200 (20992), -1.0 = 0x015000 (86016)
// relu(x) = x if positive or zero, else 0.
//
// Select DUT at iverilog time:
//   -DT27C : module GftRelu(input clk,rst_n,en, input [31:0] x, output ready, result)
//   -DT27B : module on_comb(input [31:0] x, output [31:0] t27_y)
//
// Usage: iverilog -g2012 -DT27C -o sim.vvp tb_diff_relu.v t27c_gft_relu.v
`timescale 1ns / 1ps
`default_nettype none

module tb_diff_relu;
    reg  [31:0] x;
    reg         clk = 0, rst_n = 0, en = 0;
    wire        ready;
    wire [31:0] y;
    integer     i, fails;

    `ifdef T27C
        GftRelu dut(.clk(clk), .rst_n(rst_n), .en(en), .x(x), .ready(ready), .result(y));
        localparam SAMPLING = 1;  // clocked wrapper: sample after a posedge
    `elsif T27B
        on_comb dut(.x(x), .t27_y(y));
        localparam SAMPLING = 0;  // purely combinational
    `else
        `error "define -DT27C or -DT27B"
    `endif

    always #5 clk = ~clk;

    // Vectors: {x, expected, note}
    reg [31:0] vx [0:5];
    reg [31:0] ve [0:5];
    initial begin
        vx[0] = 32'd0;     ve[0] = 32'd0;     // zero -> 0
        vx[1] = 32'd20480; ve[1] = 32'd20480; // +1.0 -> identity
        vx[2] = 32'd20992; ve[2] = 32'd20992; // +2.0 -> identity
        vx[3] = 32'd86016; ve[3] = 32'd0;     // -1.0 (bit16) -> 0
        vx[4] = 32'd131071; ve[4] = 32'd0;    // 0x1FFFF negative -> 0
        vx[5] = 32'd4095;  ve[5] = 32'd4095;  // 0x0FFF positive -> identity
    end

    task apply_step;
        begin
            x = vx[i];
            `ifdef T27C
                en = 1'b1;
                @(posedge clk); #1;
            `else
                #1;  // combinational: let deltas settle before comparing
            `endif
            if (y !== ve[i]) begin
                $display("FAIL idx=%0d x=%h got=%h exp=%h", i, x, y, ve[i]);
                fails = fails + 1;
            end else begin
                $display("PASS idx=%0d x=%h got=%h", i, x, y);
            end
        end
    endtask

    initial begin
        fails = 0;
        rst_n = 1'b0;
        #12 rst_n = 1'b1;
        for (i = 0; i < 6; i = i + 1) apply_step;
        if (fails == 0) $display("ALL_PASS");
        else $display("ALL_FAIL %0d mismatches", fails);
        $finish;
    end
endmodule
